
import json
import os


# ==========================================
# PARENT CLASS
# ==========================================

class Person:

    def __init__(self, name, age, email):
        self.name = name
        self.age = age
        self.email = email

    def display(self):
        print("Name  :", self.name)
        print("Age   :", self.age)
        print("Email :", self.email)


# ==========================================
# STUDENT CLASS (INHERITANCE)
# ==========================================

class Student(Person):

    def __init__(self, student_id, name, age, email):
        super().__init__(name, age, email)

        self.student_id = student_id
        self.enrolled_courses = []

    # Polymorphism: overriding display()
    def display(self):
        print("\n----- STUDENT DETAILS -----")
        print("Student ID :", self.student_id)
        super().display()
        print("Enrolled Courses :", self.enrolled_courses)

    def to_dict(self):
        return {
            "student_id": self.student_id,
            "name": self.name,
            "age": self.age,
            "email": self.email,
            "enrolled_courses": self.enrolled_courses
        }


# ==========================================
# COURSE CLASS
# ==========================================

class Course:

    def __init__(self, course_id, course_name,
                 instructor, duration, fee):

        self.course_id = course_id
        self.course_name = course_name
        self.instructor = instructor
        self.duration = duration
        self.fee = fee

    def display(self):
        print("\n----- COURSE DETAILS -----")
        print("Course ID   :", self.course_id)
        print("Course Name :", self.course_name)
        print("Instructor  :", self.instructor)
        print("Duration    :", self.duration)
        print("Fee         :", self.fee)

    def to_dict(self):
        return {
            "course_id": self.course_id,
            "course_name": self.course_name,
            "instructor": self.instructor,
            "duration": self.duration,
            "fee": self.fee
        }


# ==========================================
# E-LEARNING MANAGEMENT CLASS
# ==========================================

class ELearning:

    STUDENT_FILE = "students.json"
    COURSE_FILE = "courses.json"
    PROGRESS_FILE = "progress.json"

    def __init__(self):

        self.students = []
        self.courses = []
        self.progress = {}

        self.load_data()

    # ======================================
    # FILE HANDLING - LOAD DATA
    # ======================================

    def load_data(self):

        # Load students
        if os.path.exists(self.STUDENT_FILE):
            try:
                with open(self.STUDENT_FILE, "r") as file:
                    data = json.load(file)

                self.students = [
                    Student(**record) for record in data
                ]

            except (json.JSONDecodeError, TypeError, KeyError):
                print("Invalid student data file.")
                self.students = []

        # Load courses
        if os.path.exists(self.COURSE_FILE):
            try:
                with open(self.COURSE_FILE, "r") as file:
                    data = json.load(file)

                self.courses = [
                    Course(**record) for record in data
                ]

            except (json.JSONDecodeError, TypeError, KeyError):
                print("Invalid course data file.")
                self.courses = []

        # Load progress
        if os.path.exists(self.PROGRESS_FILE):
            try:
                with open(self.PROGRESS_FILE, "r") as file:
                    self.progress = json.load(file)

            except json.JSONDecodeError:
                print("Invalid progress data file.")
                self.progress = {}

    # ======================================
    # FILE HANDLING - SAVE DATA
    # ======================================

    def save_students(self):

        with open(self.STUDENT_FILE, "w") as file:
            json.dump(
                [student.to_dict() for student in self.students],
                file,
                indent=4
            )

    def save_courses(self):

        with open(self.COURSE_FILE, "w") as file:
            json.dump(
                [course.to_dict() for course in self.courses],
                file,
                indent=4
            )

    def save_progress(self):

        with open(self.PROGRESS_FILE, "w") as file:
            json.dump(self.progress, file, indent=4)

    # ======================================
    # ADD NEW COURSE
    # ======================================

    def add_course(self):

        print("\n--- ADD NEW COURSE ---")

        course_id = input("Enter Course ID: ")

        if any(c.course_id == course_id for c in self.courses):
            print("Course ID already exists!")
            return

        course_name = input("Enter Course Name: ")
        instructor = input("Enter Instructor Name: ")
        duration = input("Enter Course Duration: ")

        try:
            fee = float(input("Enter Course Fee: "))
            if fee < 0:
                print("Fee cannot be negative.")
                return
        except ValueError:
            print("Invalid course fee!")
            return

        course = Course(
            course_id, course_name,
            instructor, duration, fee
        )

        self.courses.append(course)
        self.save_courses()

        print("Course added successfully!")

    # ======================================
    # DISPLAY ALL COURSES
    # ======================================

    def display_courses(self):

        print("\n--- AVAILABLE COURSES ---")

        if not self.courses:
            print("No courses available.")
            return

        for course in self.courses:
            course.display()

    # ======================================
    # REGISTER NEW STUDENT
    # ======================================

    def register_student(self):

        print("\n--- STUDENT REGISTRATION ---")

        student_id = input("Enter Student ID: ")

        if any(s.student_id == student_id
               for s in self.students):
            print("Student ID already exists!")
            return

        name = input("Enter Student Name: ")

        try:
            age = int(input("Enter Age: "))
            if age <= 0:
                print("Age must be positive.")
                return
        except ValueError:
            print("Invalid age!")
            return

        email = input("Enter Email: ")

        student = Student(
            student_id, name, age, email
        )

        self.students.append(student)
        self.save_students()

        print("Student registered successfully!")

    # ======================================
    # DISPLAY ALL STUDENTS
    # ======================================

    def display_students(self):

        print("\n--- REGISTERED STUDENTS ---")

        if not self.students:
            print("No students registered.")
            return

        for student in self.students:
            student.display()

    # ======================================
    # SEARCH COURSE
    # ======================================

    def search_course(self):

        course_id = input("Enter Course ID: ")

        for course in self.courses:

            if course.course_id == course_id:
                course.display()
                return

        print("Course not found!")

    # ======================================
    # ENROLL STUDENT IN COURSE
    # ======================================

    def enroll_student(self):

        print("\n--- COURSE ENROLLMENT ---")

        student_id = input("Enter Student ID: ")
        course_id = input("Enter Course ID: ")

        student = next(
            (s for s in self.students
             if s.student_id == student_id),
            None
        )

        course = next(
            (c for c in self.courses
             if c.course_id == course_id),
            None
        )

        if student is None:
            print("Student not found!")
            return

        if course is None:
            print("Course not found!")
            return

        if course_id in student.enrolled_courses:
            print("Student already enrolled!")
            return

        student.enrolled_courses.append(course_id)

        # Initialize course progress
        self.progress.setdefault(student_id, {})
        self.progress[student_id][course_id] = 0

        self.save_students()
        self.save_progress()

        print("Student enrolled successfully!")

    # ======================================
    # UPDATE COURSE PROGRESS
    # ======================================

    def update_progress(self):

        print("\n--- UPDATE COURSE PROGRESS ---")

        student_id = input("Enter Student ID: ")
        course_id = input("Enter Course ID: ")

        student = next(
            (s for s in self.students
             if s.student_id == student_id),
            None
        )

        if student is None:
            print("Student not found!")
            return

        if course_id not in student.enrolled_courses:
            print("Student is not enrolled in this course!")
            return

        try:
            percentage = int(
                input("Enter Progress (0-100): ")
            )

            if not 0 <= percentage <= 100:
                print("Progress must be between 0 and 100.")
                return

        except ValueError:
            print("Invalid progress value!")
            return

        self.progress.setdefault(student_id, {})
        self.progress[student_id][course_id] = percentage

        self.save_progress()

        print("Course progress updated successfully!")

    # ======================================
    # VIEW STUDENT PROGRESS
    # ======================================

    def view_progress(self):

        student_id = input("Enter Student ID: ")

        student = next(
            (s for s in self.students
             if s.student_id == student_id),
            None
        )

        if student is None:
            print("Student not found!")
            return

        print("\n--- STUDENT LEARNING PROGRESS ---")
        print("Student Name:", student.name)

        if not student.enrolled_courses:
            print("No courses enrolled.")
            return

        for course_id in student.enrolled_courses:

            percentage = self.progress.get(
                student_id, {}
            ).get(course_id, 0)

            course = next(
                (c for c in self.courses
                 if c.course_id == course_id),
                None
            )

            course_name = (
                course.course_name if course
                else "Unknown Course"
            )

            print(
                course_name, ":", percentage, "%"
            )

    # ======================================
    # UPDATE COURSE DETAILS
    # ======================================

    def update_course(self):

        course_id = input("Enter Course ID: ")

        course = next(
            (c for c in self.courses
             if c.course_id == course_id),
            None
        )

        if course is None:
            print("Course not found!")
            return

        print("Press Enter to keep existing values.")

        name = input("New Course Name: ")
        instructor = input("New Instructor: ")
        duration = input("New Duration: ")
        fee = input("New Fee: ")

        if name:
            course.course_name = name

        if instructor:
            course.instructor = instructor

        if duration:
            course.duration = duration

        if fee:
            try:
                new_fee = float(fee)
                if new_fee < 0:
                    print("Fee cannot be negative.")
                    return
                course.fee = new_fee
            except ValueError:
                print("Invalid fee!")
                return

        self.save_courses()

        print("Course updated successfully!")

    # ======================================
    # DELETE COURSE
    # ======================================

    def delete_course(self):

        course_id = input("Enter Course ID: ")

        course = next(
            (c for c in self.courses
             if c.course_id == course_id),
            None
        )

        if course is None:
            print("Course not found!")
            return

        confirm = input(
            "Are you sure? (yes/no): "
        )

        if confirm.lower() == "yes":

            self.courses.remove(course)

            # Remove course from all students
            for student in self.students:
                if course_id in student.enrolled_courses:
                    student.enrolled_courses.remove(course_id)

            # Remove saved progress
            for student_id in self.progress:
                self.progress[student_id].pop(
                    course_id, None
                )

            self.save_courses()
            self.save_students()
            self.save_progress()

            print("Course deleted successfully!")

        else:
            print("Deletion cancelled.")


# ==========================================
# MAIN PROGRAM
# ==========================================

def main():

    system = ELearning()

    while True:

        print("\n======================================")
        print("     E-LEARNING COURSE MANAGEMENT")
        print("======================================")

        print("1. Add Course")
        print("2. Display All Courses")
        print("3. Register Student")
        print("4. Display Students")
        print("5. Search Course")
        print("6. Enroll Student")
        print("7. Update Course Progress")
        print("8. View Student Progress")
        print("9. Update Course")
        print("10. Delete Course")
        print("11. Exit")

        choice = input("\nEnter Your Choice: ")

        if choice == "1":
            system.add_course()

        elif choice == "2":
            system.display_courses()

        elif choice == "3":
            system.register_student()

        elif choice == "4":
            system.display_students()

        elif choice == "5":
            system.search_course()

        elif choice == "6":
            system.enroll_student()

        elif choice == "7":
            system.update_progress()

        elif choice == "8":
            system.view_progress()

        elif choice == "9":
            system.update_course()

        elif choice == "10":
            system.delete_course()

        elif choice == "11":
            print("\nThank you for using E-Learning System!")
            break

        else:
            print("Invalid choice! Please try again.")


# Execute Program
if __name__ == "__main__":
    main()
